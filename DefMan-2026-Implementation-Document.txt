DROP TABLE IF EXISTS `user_sessions`;

  `id` SMALLINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(100) NOT NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

-- -----------------------------------------------------------------------------
-- 2. Table: `users` (Refactored from install/installdb.php lines 55-82)
-- User Levels:
--   0 = Applying (Pending Admin Approval - form_apply.php)
--   1 = Public   (Unauthenticated / Public reporter - form_raportpub.php)
--   2 = Customer (Authenticated defect reporter - form_raport.php)

CREATE TABLE `users` (
  `id` SMALLINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `login` VARCHAR(64) NOT NULL COMMENT 'Legacy max 6 chars in 2004 UI; widened for modern auth',
  `password` VARCHAR(255) NOT NULL COMMENT 'Argon2id password hash (PASSWORD_ARGON2ID)',
  `email` VARCHAR(255) NOT NULL,
  `userlevel` TINYINT UNSIGNED NOT NULL DEFAULT 0 COMMENT '0=Applying, 1=Public, 2=Customer, 3=Analyzer, 4=Admin',
  `company` VARCHAR(100) DEFAULT NULL,
  `phone` VARCHAR(50) DEFAULT NULL,
  `name` VARCHAR(100) DEFAULT NULL,
  `activate` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '0=Pending/Inactive, 1=Active',
  `logged` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '0=Offline, 1=Online',
  `reset_token_hash` CHAR(64) DEFAULT NULL COMMENT 'SHA-256 hash of random_bytes(32) recovery token',
  `reset_token_expires_at` DATETIME DEFAULT NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_users_login` (`login`),
  UNIQUE KEY `uq_users_email` (`email`),
  KEY `idx_users_level_activate` (`userlevel`, `activate`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- -----------------------------------------------------------------------------
-- 3. Table: `user_sessions` (Cryptographically secure bearer/session tokens)

CREATE TABLE `user_sessions` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `user_id` SMALLINT UNSIGNED NOT NULL,
  `token_hash` CHAR(64) NOT NULL COMMENT 'SHA-256 hash of 256-bit random_bytes token',
  `ip_address` VARCHAR(45) DEFAULT NULL,
  `user_agent` VARCHAR(255) DEFAULT NULL,
  `expires_at` DATETIME NOT NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_user_sessions_token_hash` (`token_hash`),
  KEY `idx_user_sessions_user_id` (`user_id`),
  KEY `idx_user_sessions_expires_at` (`expires_at`),
  CONSTRAINT `fk_user_sessions_user`
    FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/**
 * IEEE (1992) Standard Classification for Software Anomalies (IEEE Std 1044-1993 / 1992 draft)
 * Refactored from: defecttype.txt & faulttypes.html
 */
enum IeeeFaultType: string
{
    case LOGIC                  = 'Logic';
    case COMPUTATIONAL          = 'Computational';
    case INTERFACE              = 'Interface';
    case DATA_HANDLING          = 'Data handling';
    case DATA                   = 'Data';
    case DOCUMENTATION          = 'Documentation';
    case DOCUMENT_QUALITY       = 'Document quality';
    case ENHANCEMENT            = 'Enhancement';
    case CAUSED_BY_PREVIOUS_FIX = 'Caused by a prev. Fix';
    case OTHER                  = 'Other';

    /**
     * Hex color palette used by PixiJS v8 WebGPU DefectTriageCanvas & UI badges
     */
    public function hexColor(): int
    {
        return match ($this) {
            self::LOGIC                  => 0xEF4444, // Red-500
            self::COMPUTATIONAL          => 0xF97316, // Orange-500
            self::INTERFACE              => 0xEC4899, // Pink-500
            self::DATA_HANDLING          => 0x8B5CF6, // Violet-500

                'Other problem',
            ],
        };
    }

    /**
     * Normalizes legacy case variations (e.g. "Caused by a prev. fix" vs "Caused by a prev. Fix")
     */
    public static function fromLegacyString(string $value): self
    {
        $trimmed = trim($value);
        foreach (self::cases() as $case) {
            if (strcasecmp($case->value, $trimmed) === 0) {
                return $case;
            }
        }
        return self::OTHER;
    }

<?php
declare(strict_types=1);

namespace DefMan\Infrastructure\Database;

use PDO;
use PDOException;
use RuntimeException;

/**
 * Strict-typed PHP 8.3 PDO Connection Singleton
 * Refactored from: config.php & dbconnect.php
 */
final class Database
{
    private static ?PDO $instance = null;

    private function __construct() {}

    private function __clone() {}

    public function __wakeup(): never
    {
        throw new RuntimeException('Cannot unserialize singleton Database instance.');
    }

    public static function getConnection(): PDO
    {
        if (self::$instance === null) {
            $host    = $_ENV['DB_HOST'] ?? getenv('DB_HOST') ?: 'localhost';
            $port    = (int) ($_ENV['DB_PORT'] ?? getenv('DB_PORT') ?: 3306);
            $dbName  = $_ENV['DB_NAME'] ?? getenv('DB_NAME') ?: 'opr0413';
            $user    = $_ENV['DB_USER'] ?? getenv('DB_USER') ?: 'opr0413';
            $pass    = $_ENV['DB_PASS'] ?? getenv('DB_PASS') ?: 'd3fman13';
            $charset = 'utf8mb4';

            $dsn = sprintf('mysql:host=%s;port=%d;dbname=%s;charset=%s', $host, $port, $dbName, $charset);

            try {
                self::$instance = new PDO($dsn, $user, $pass, [
                    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
                    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                    PDO::ATTR_EMULATE_PREPARES   => false,
                    PDO::ATTR_STRINGIFY_FETCHES  => false,
                ]);
            } catch (PDOException $e) {
                throw new RuntimeException('Database connection failed: ' . $e->getMessage(), (int) $e->getCode(), $e);
            }
        }

        return self::$instance;
    }
}

     * Multi-column sorting & filtering by severity, priority, status, product, and customer.
     * Also computes `days_since_submitted` and deleted-user badge per form_raportlist.php.
     */
    public function findFiltered(
        ?string $severity = null,
        ?string $priority = null,
        ?string $status = null,
        ?string $product = null,
        ?int $customerId = null,
        string $sortBy = 'date'
    ): array {
        $conditions = [];
        $params     = [];

        if ($severity !== null && $severity !== '') {
            $conditions[]        = 'r.`Severity` = :severity';
            $params[':severity'] = $severity;
        }

        if ($priority !== null && $priority !== '') {
            $normalizedPriority  = DefectPriority::fromInput($priority)->value;
            $conditions[]        = 'r.`Priority` = :priority';
            $params[':priority'] = $normalizedPriority;
        }

        if ($status !== null && $status !== '') {
            if ($status === 'unassigned') {
                $conditions[] = 'a.`status` IS NULL';
            } else {
                $conditions[]      = 'a.`status` = :status';
                $params[':status'] = $status;
            }
        }

        if ($product !== null && $product !== '') {
            $conditions[]       = 'r.`Product` = :product';
            $params[':product'] = $product;
        }

        if ($customerId !== null) {
            $conditions[]           = 'r.`CustomerID` = :customer_id';
            $params[':customer_id'] = $customerId;
        }

        $whereSql = count($conditions) > 0 ? 'WHERE ' . implode(' AND ', $conditions) : '';

        $orderBySql = match ($sortBy) {
            'product'  => 'r.`Product` ASC, r.`ID` DESC',
            'contact'  => 'r.`Who` DESC, r.`ID` DESC',
            'severity' => "FIELD(r.`Severity`, 'critical', 'major', 'minor') ASC, r.`Submitted` DESC",
            'priority', 'all' => "FIELD(r.`Priority`, 'm_urgent', 'average', 'l_urgent') ASC, r.`Submitted` DESC",
            default    => 'r.`Submitted` DESC, r.`ID` DESC',
        };

        $sql = "
            SELECT
                r.`ID`,
                r.`Product`,
                r.`product_id`,
                r.`Environment`,
                r.`CustomerID`,
                r.`Who`,
                r.`Contactemail`,
                r.`Whenreported`,
                r.`Title`,
                r.`Description`,
                r.`Discoveredby`,
                r.`Persontype`,
                r.`Occured`,
                r.`Location`,
                r.`Severity`,
                r.`Priority`,
                CASE r.`Priority`
                    WHEN 'm_urgent' THEN 'most urgent'
                    WHEN 'l_urgent' THEN 'less urgent'
                    ELSE 'average'
                END AS `PriorityLabel`,
                r.`Submitted`,
                DATEDIFF(CURDATE(), r.`Submitted`) AS `days_since_submitted`,
                r.`Addinfo`,
                CASE WHEN r.`CustomerID` IS NOT NULL AND u.`id` IS NULL THEN 1 ELSE 0 END AS `is_deleted_user`,
                a.`classification`,
                COALESCE(a.`status`, 'submitted') AS `analysis_status`,
                a.`severity` AS `developer_severity`,
                a.`priority` AS `developer_priority`,
                a.`fixeffort`,
                a.`fixcost`,
                a.`costofdefect`,
                a.`assigned`,
                a.`fixcomplete`,
                a.`resocomplete`,
                a.`resoclosed`
            FROM `raport` r
            LEFT JOIN `users` u ON u.`id` = r.`CustomerID`
            LEFT JOIN `analyse` a ON a.`id` = r.`ID`
            {$whereSql}
            ORDER BY {$orderBySql}
        ";

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt->fetchAll();
    }

<?php
declare(strict_types=1);

namespace DefMan\Http\Middleware;

use DefMan\Domain\Enum\UserLevel;
use DefMan\Repository\UserRepository;

/**
 * RBAC Authentication & Authorization Middleware
 * Enforces the 4 DefMan user levels:
 *   - Level 1 (Public):   Unauthenticated / public defect submission
 *   - Level 2 (Customer): Authenticated user tracking their own submitted defects
 *   - Level 3 (Analyzer): Developer/QA triage, IEEE fault classification, fix & cost tracking
 *   - Level 4 (Admin):    User registration approval queue, role assignment, product management
 */
final readonly class RbacMiddleware
{
    public function __construct(private UserRepository $userRepository) {}

    public function extractBearerToken(): ?string
    {
        $header = $_SERVER['HTTP_AUTHORIZATION'] ?? $_SERVER['REDIRECT_HTTP_AUTHORIZATION'] ?? '';
        if (preg_match('/^Bearer\s+([a-f0-9]{64})$/i', $header, $matches)) {
            return $matches[1];
        }
        return $_COOKIE['defman_token'] ?? null;
    }

    public function authenticateOptional(): ?array
    {
        $token = $this->extractBearerToken();
        if ($token === null) {
            return null;
        }
        return $this->userRepository->findUserBySessionToken($token);
    }

    public function requireLevel(UserLevel $minimumLevel): array
    {
        if ($minimumLevel === UserLevel::PUBLIC) {
            return $this->authenticateOptional() ?? [
                'id'        => null,
                'login'     => 'public',
                'userlevel' => UserLevel::PUBLIC->value,
            ];
        }

        $user = $this->authenticateOptional();
        if ($user === null) {
            http_response_code(401);
            header('Content-Type: application/json');
            echo json_encode(['error' => 'Authentication required.']);
            exit;
        }

        $userLevel = (int) $user['userlevel'];
        if ($userLevel < $minimumLevel->value) {
            http_response_code(403);
            header('Content-Type: application/json');
            echo json_encode([
                'error'          => 'Insufficient permissions.',
                'required_level' => $minimumLevel->label(),
                'current_level'  => UserLevel::tryFrom($userLevel)?->label() ?? 'Unknown',
            ]);
            exit;
        }

        return $user;
    }
}

spl_autoload_register(static function (string $class): void {
    $prefix = 'DefMan\\';
    if (!str_starts_with($class, $prefix)) {
        return;
    }
    $relative = substr($class, strlen($prefix));
    $file = __DIR__ . '/../src/' . str_replace('\\', '/', $relative) . '.php';
    if (is_file($file)) {
        require_once $file;
    }
});

header('Access-Control-Allow-Origin: *');
header('Access-Control-Allow-Headers: Content-Type, Authorization');
header('Access-Control-Allow-Methods: GET, POST, PUT, PATCH, DELETE, OPTIONS');

if ($_SERVER['REQUEST_METHOD'] === 'OPTIONS') {

    $pdo            = Database::getConnection();
    $userRepo       = new UserRepository($pdo);

    $defectCtrl     = new DefectController($defectRepo, $analysisRepo, $rbac);
    $productCtrl    = new ProductController($productRepo, $rbac);

    $method = $_SERVER['REQUEST_METHOD'];
    $uri    = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH) ?: '/';
    $rawBody = file_get_contents('php://input');
    $body   = $rawBody ? (json_decode($rawBody, true) ?? []) : [];

    $response = match (true) {
        // Auth & Registration
        $method === 'POST' && $uri === '/api/auth/login'         => $authCtrl->login($body),
        $method === 'POST' && $uri === '/api/auth/logout'        => $authCtrl->logout(),
        $method === 'POST' && $uri === '/api/auth/apply'         => $authCtrl->apply($body),
        $method === 'POST' && $uri === '/api/auth/lost-password' => $authCtrl->lostPassword($body),

        // User Profile (Level 2+)
        $method === 'GET'  && $uri === '/api/users/me'           => $userCtrl->getMe(),
        $method === 'PUT'  && $uri === '/api/users/me'           => $userCtrl->updateMe($body),

        // Admin User Approval & Management (Level 4)
        $method === 'GET'    && $uri === '/api/admin/users'      => $userCtrl->listUsers(),
        $method === 'POST'   && $uri === '/api/admin/users'      => $userCtrl->createUser($body),
        $method === 'PATCH'  && preg_match('#^/api/admin/users/(\d+)$#', $uri, $m) === 1 => $userCtrl->updateUserStatus((int) $m[1], $body),
        $method === 'DELETE' && preg_match('#^/api/admin/users/(\d+)$#', $uri, $m) === 1 => $userCtrl->deleteUser((int) $m[1]),

        // Products (Public GET, Level 4 Admin POST/DELETE)
        $method === 'GET'    && ($uri === '/api/products' || $uri === '/api/admin/products') => $productCtrl->listProducts(),
        $method === 'POST'   && $uri === '/api/admin/products'   => $productCtrl->createProduct($body),
        $method === 'DELETE' && preg_match('#^/api/admin/products/(\d+)$#', $uri, $m) === 1 => $productCtrl->deleteProduct((int) $m[1]),

        // Public & Customer Defect Reporting
        $method === 'POST'   && $uri === '/api/defects/public'   => $defectCtrl->createPublicReport($body),
        $method === 'POST'   && $uri === '/api/defects'          => $defectCtrl->createCustomerReport($body),
        $method === 'GET'    && $uri === '/api/defects'          => $defectCtrl->listDefects($_GET),
        $method === 'GET'    && preg_match('#^/api/defects/(\d+)$#', $uri, $m) === 1 => $defectCtrl->getDefect((int) $m[1]),
        $method === 'PUT'    && preg_match('#^/api/defects/(\d+)$#', $uri, $m) === 1 => $defectCtrl->updateDefectReport((int) $m[1], $body),
        $method === 'DELETE' && preg_match('#^/api/defects/(\d+)$#', $uri, $m) === 1 => $defectCtrl->deleteDefect((int) $m[1]),

        // Analyzer Triage, IEEE 1992 Classification & Cost Resolution (Level 3+)
        $method === 'GET'    && $uri === '/api/analyzer/queue'   => $defectCtrl->getAnalyzerQueue($_GET),
        $method === 'PUT'    && preg_match('#^/api/defects/(\d+)/analyze$#', $uri, $m) === 1 => $defectCtrl->analyzeDefect((int) $m[1], $body),
        $method === 'PUT'    && preg_match('#^/api/defects/(\d+)/resolve$#', $uri, $m) === 1 => $defectCtrl->resolveDefect((int) $m[1], $body),

        default => (static function (): array {
            http_response_code(404);
            return ['error' => 'Endpoint not found.'];
        })(),
    };

    echo json_encode($response, JSON_THROW_ON_ERROR | JSON_UNESCAPED_UNICODE);
} catch (\Throwable $e) {
    http_response_code(500);
    echo json_encode([
        'error'   => 'Internal Server Error',
        'details' => $e->getMessage(),
    ]);
}

const API_BASE = import.meta.env?.VITE_API_BASE_URL ?? 'http://localhost:8080';

async function request<T>(path: string, init: RequestInit = {}): Promise<T> {
  const token = localStorage.getItem('defman_token');
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...(init.headers as Record<string, string> | undefined),
  };
  if (token) {
    headers['Authorization'] = `Bearer ${token}`;
  }

  const res = await fetch(`${API_BASE}${path}`, {
    ...init,
    headers,
  });

  const data = await res.json();
  if (!res.ok) {
    const message =
      data.error ||
      (Array.isArray(data.errors) ? data.errors.join(' - ') : 'Request failed');
    throw new Error(message);
  }
  return data as T;
}

  return { color: 0x38bdf8, alpha: 0.45, width: 2 };
}

function computeNodeRadius(effortHours: number, monetaryCostEur: number): number {
  const effortComponent = Math.sqrt(Math.max(1, effortHours)) * 4.5;
  const costComponent = Math.log10(Math.max(10, monetaryCostEur + 10)) * 7;
  return Math.min(54, Math.max(20, Math.round(12 + effortComponent + costComponent)));
}

          const dy = currentLocal.y - dragStartPos.y;
          pointerMovedDistance += Math.hypot(dx, dy);
          nodeContainer.position.x += dx;
          nodeContainer.position.y += dy;
          dragStartPos = currentLocal;

        world.position.y += e.global.y - panLast.y;

        stagePanning = false;

        setZoomLevel(Number(nextScale.toFixed(2)));
      };

      app.canvas.addEventListener('wheel', onWheel, { passive: false });
    };

      destroyed = true;
      try {
        app.destroy(true, { children: true });
      } catch {
        // Safe cleanup if unmount occurs mid-init
      }
    };

          <p className="text-xs text-slate-400">
            Drag nodes across lanes (Open → Analyzing → Fixed → Resolved) to trigger{' '}
            <code className="text-amber-300">PUT /api/defects/&#123;id&#125;/resolve</code>, or click any node to open{' '}
            <code className="text-amber-300">&lt;DefectAnalyzerWorkbench /&gt;</code>.

        </div>
        <div className="flex items-center gap-3 text-xs">

            {rendererBackend}
